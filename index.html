Setup en 4 étapes :
  1. Créer un Google Sheet → renommer la 1ère feuille data
  2. Menu Extensions → Apps Script → coller le code ci-dessous → remplacer SHEET_ID par l'ID de votre feuille (dans l'URL entre /d/ et /edit)
  3. Déployer → Nouveau déploiement → Type: Application Web → Accès: Tout le monde → Copier l'URL
  4. Coller l'URL ici et cliquer Tester
Code Apps Script : 
const SHEET_ID = "VOTRE_ID_ICI";

function doPost(e) {
  const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName("data");
  const p = JSON.parse(e.postData.contents);
  if (p.action==="getAll") {
    const d=sheet.getDataRange().getValues();
    const r={};d.forEach(row=>{if(row[0])r[row[0]]=row[1];});
    return ContentService.createTextOutput(JSON.stringify(r)).setMimeType(ContentService.MimeType.JSON);
  }
  if (p.action==="set") {
    const d=sheet.getDataRange().getValues();
    let f=false;
    for(let i=0;i<d.length;i++){if(d[i][0]===p.key){sheet.getRange(i+1,2).setValue(p.value);f=true;break;}}
    if(!f)sheet.appendRow([p.key,p.value]);
    return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON);
  }
  if (p.action==="setBatch") {
    const d=sheet.getDataRange().getValues();
    const m={};d.forEach((r,i)=>{m[r[0]]=i;});
    p.entries.forEach(({key,value})=>{
      if(key in m)sheet.getRange(m[key]+1,2).setValue(value);
      else sheet.appendRow([key,value]);
    });
    return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({error:"?"})).setMimeType(ContentService.MimeType.JSON);
}
function doGet(e){return doPost({postData:{contents:JSON.stringify({action:"getAll"})}});}
